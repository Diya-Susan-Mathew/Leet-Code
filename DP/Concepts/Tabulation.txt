public class Tabulation {

    public static long fibonacci(int n) {
        // Base cases: if n is 0 or 1, we just return n.
        if (n <= 1) {
            return n;
        }

        // Create an array to store our past answers (the "Tabulation" table)
        // Size is n + 1 so we can store up to the nth index.
        long[] dp = new long[n + 1];

        // Seed the array with the known base cases
        dp[0] = 0;
        dp[1] = 1;

        // Build the sequence from the bottom up!
        for (int i = 2; i <= n; i++) {
            // The current number is just the sum of the two previous answers we saved.
            dp[i] = dp[i - 1] + dp[i - 2];
        }

        // The answer to our ultimate problem is sitting at the very end of the array.
        return dp[n];
    }

    public static void main(String[] args) {
        int n = 50; 
        System.out.println("Fibonacci of " + n + " is: " + fibonacci(n));
        // This will run instantly thanks to DP!
    }
}
