class Solution {
    public boolean find132pattern(int[] nums) {

        for(int i=0;i<nums.length-2;i++){
            int j = i+1;
            int k = i+2;
            if((nums[i]<nums[j]) && (nums[j]>nums[k])){
                return true;
            }
            if((nums[i]<nums[nums.length-2]) && (nums[nums.length-2]>nums[nums.length-1])){
                return true;
            }
        }
        return false;

    }
}
