Given an array arr of integers, check if there exist two indices i and j such that :

i != j
0 <= i, j < arr.length
arr[i] == 2 * arr[j]
 

Example 1:

Input: arr = [10,2,5,3]
Output: true
Explanation: For i = 0 and j = 2, arr[i] == 10 == 2 * 5 == 2 * arr[j]




// My Bruteforce but corrected by gemini
import java.util.Arrays;
class Solution {
    public boolean checkIfExist(int[] arr) {

        // Built-in sort is much faster O(n log n) than Bubble Sort O(n²)
        Arrays.sort(arr); 
        
        for (int i = 0; i < arr.length; i++) {
            int target = arr[i] * 2;
            int left = 0;
            int right = arr.length - 1;
            
            // Binary search for the target
            while (left <= right) {
                int mid = left + (right - left) / 2;
                
                if (arr[mid] == target && mid != i) { // Ensure it's not the same element
                    return true;
                } else if (arr[mid] < target) {
                    left = mid + 1;
                } else {
                    right = mid - 1;
                }
            }
        }
        return false;
    }
}
